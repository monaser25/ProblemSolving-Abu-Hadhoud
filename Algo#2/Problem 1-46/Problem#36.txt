#include <iostream>   
using namespace std;  


int ReadNumber()
{
    int Number;  
    cout << "\nPlease enter a number \n";
    cin >> Number; 
    return Number;   
}

void AddArrayElement(int Number,int arr[],int &length)
{
    length++;
    arr[length - 1] = Number;
}
void InputUserNumbersInArray(int arr[],int &length)
{
    bool AddMore = true;
    do
    {
        AddArrayElement(ReadNumber(), arr, length);
        cout << "\n Do You Want To Add More Number ? [0]:No [1]:Yes ";
        cin >> AddMore;
    } while (AddMore);
}
void PrintArray(int arr[100], int arrLength)
{
    
    for (int i = 0; i < arrLength; i++)
        cout << arr[i] << " ";
    cout << "\n";  
    cout << arrLength;
}



int main()
{
    srand((unsigned)time(NULL));

    int arr[100];   
    int arrLength = 0;
    InputUserNumbersInArray(arr, arrLength);
    PrintArray(arr, arrLength);

  


    return 0;  
}
