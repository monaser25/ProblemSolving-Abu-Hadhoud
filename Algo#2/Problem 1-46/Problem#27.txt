#include <iostream>

using namespace std;

int RandomNumbers(int From, int To)
{
	int randnumber =rand()%To - From + 1 + From;
	return randnumber;
}
void FillArrayWithRandomNumbers(int arr[], int& length)
{
	cout << "Enter Number of Elemnets " << endl;
	cin >> length;
	for (int i = 0; i < length; i++)
	{
		arr[i] = RandomNumbers(1, 100);
	}
}
void PrintArray(int arr[],int length)
{

	for (int i = 0; i < length; i++)
	{
		cout << arr[i] << " ";
	}
	cout << endl;
}

int SumNumberInArray(int arr[100], int length)
{
	int sum = 0;

	for (int i = 0; i < length; i++)
	{
		
			sum += arr[i];
		
	}

	return sum;
}

int main()
{
	srand((unsigned)time(NULL));
	int arr[100], length, sum;

	FillArrayWithRandomNumbers(arr, length);
	cout << "Array Is " << endl;
	PrintArray(arr,length);

	sum=SumNumberInArray(arr, length);
	cout << "Sum is " << sum << endl;
	cout << "Avg is " << sum / length;
	return 0;
}
